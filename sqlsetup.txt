RETURNS text

IMMUTABLE
AS $$
  SELECT lower(trim('@umatillacounty.gov'))::text;

-- === 6. FUNCTIONS ===
CREATE OR REPLACE FUNCTION public.delete_empty_map_groups()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
  -- Check if the old group ID still exists in the map_groups table
  IF OLD.group_id IS NOT NULL THEN
    -- Delete the group if no maps are currently assigned to it
    DELETE FROM map_groups
    WHERE id = OLD.group_id
    AND NOT EXISTS (
      SELECT 1 FROM maps_registry WHERE group_id = OLD.group_id
    );
  END IF;
  RETURN OLD;
END;
$function$
;

-- === 7. TRIGGERS ===
DROP TRIGGER IF EXISTS trigger_cleanup_empty_groups ON public.maps_registry;
CREATE TRIGGER trigger_cleanup_empty_groups AFTER DELETE OR UPDATE OF group_id ON maps_registry FOR EACH ROW EXECUTE FUNCTION delete_empty_map_groups();
